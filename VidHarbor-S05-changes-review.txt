+from django.db import transaction
+from django.http import JsonResponse
+from django.shortcuts import get_object_or_404
+from django.views.decorators.cache import never_cache
+from django.views.decorators.http import require_POST
+
+from .accounts import error, json_object, require_author
+from .models import Video
+
+
+@require_POST
+@never_cache
+@require_author
+def rename(request, video_id):
+    with transaction.atomic():
+        video = get_object_or_404(
+            Video.objects.select_for_update(), id=video_id, owner=request.user,
+        )
+        title = json_object(request).get("title")
+        if not isinstance(title, str) or not 1 <= len(title.strip()) <= 200 or "\0" in title:
+            return error("Введите название длиной от 1 до 200 символов.", 400)
+        video.title = title.strip()
+        video.save(update_fields=["title"])
+    return JsonResponse({"id": video.id, "title": video.title, "state": video.state})

+# S05 · Управление своими видео
+
+**Статус · План для согласования.** S03 уже в `main` (merge `f120ac2`). Для сквозных проверок S05 нужен результат S04. Сейчас можно выполнить переименование в кабинете и проверки прав владельца; повтор обработки, удаление при живом FFmpeg и отзыв HLS подключаются к фактическому коду S04.
+
+Основание: [T501–T503](../tasks.md#s05--управление-своими-видео), [FR-008, FR-010 и AC-005–006](product-spec.md), [контракты надёжности](homework-05/reliability.md).
+
+## Порядок и границы
+
+1. Переименование через API и кабинет, права владельца, CSRF, валидация и сохранение идентичности видео. Эта часть доступна на базе S03.
+2. После S04 — интеграция нового названия с каталогом, поиском и страницей; ручной повтор из оригинала с новым поколением.
+3. После S04 — логическое удаление, отзыв выдачи, остановка FFmpeg и повторяемая очистка под общей блокировкой S04.
+4. Проверки гонок, перезапуска и устойчивой отметки удаления.
+5. После T602 — внешний сценарий отмены уведомления. В T802 — проверка журнала удалений при восстановлении старой резервной копии.
+
+Проверки с остановкой процессов и удалением проводить в отдельном проекте Compose с тестовыми томами. Не закрывать весь T501 только по результату переименования и весь T503 до зависимых сценариев.

+
+- Добавить команду переименования с проверкой сессии, CSRF и владельца. Для чужого и отсутствующего ID возвращать одинаковый отказ. Принимать непустое строковое название до 200 символов после обрезки пробелов, отклонять NUL.
+- Менять только название Video. Сохранить ID, ссылку, поколение, оригинал, задания и outbox. Название первоначальной загрузки и снимки уже созданных событий не переписывать.
+- Добавить редактирование, сохранение, отмену, состояние ожидания и ошибку в кабинете. Сохранять введённый текст при ошибке. Выводить название текстом, без исполнения HTML.

+## T501 · Ручной повтор из оригинала
+
+- Разрешать повтор владельцу после окончательного устранимого сбоя. Отказ для повреждённого или неподдерживаемого оригинала объясняет необходимость загрузить исправный файл. Не принимать повтор для активного, готового или удаляемого видео.
+- **Привязать запрос к конкретному неудачному job_id**, а не только к video_id и текущему состоянию. Сохранить устойчивую связь исходного неудачного задания с созданным заданием следующего поколения; защитить связь уникальным ограничением БД.
+- После проверки владельца сначала искать уже принятый повтор для этого исходного job_id. class RenameVideoTests(TestCase):
    def setUp(self):
        self.owner = get_user_model().objects.create_user(username="owner")
        self.other = get_user_model().objects.create_user(username="other")
        self.upload = Upload.objects.create(
            owner=self.owner, title="Оригинал", expected_size=123, size=123,
            sha256="a" * 64, state=Upload.State.COMPLETE,
        )
        self.video = Video.objects.create(owner=self.owner, upload=self.upload, title="Прежнее")
        self.job = Job.objects.create(video=self.video, generation=self.video.generation)
        self.outbox = Outbox.objects.create(
            job=self.job, kind="video.transcode.v1", attempt_no=1, payload={"title": "Прежнее"},
        )
        self.url = f"/api/videos/{self.video.id}/rename"
        self.client = Client(enforce_csrf_checks=True)
        self.login(self.owner)

    def login(self, user):
        self.client.force_login(user)
        token = self.client.get("/api/auth/session").json()["csrf_token"]
        self.headers = {"X-CSRFToken": token}

    def rename(self, title, url=None):
        return self.client.post(
            url or self.url, {"title": title}, content_type="application/json", headers=self.headers,
        )

    def test_rename_preserves_identity_original_and_processing(self):
        before_video = Video.objects.values().get(pk=self.video.pk)
        before_upload = Upload.objects.values().get(pk=self.upload.pk)
        before_job = Job.objects.values().get(pk=self.job.pk)
        before_outbox = Outbox.objects.values().get(pk=self.outbox.pk)
        for _ in range(2):
            response = self.rename("  Новое название  ")
            self.assertEqual(response.status_code, 200)
            self.assertEqual(response.json()["id"], str(self.video.pk))
            self.assertEqual(response.json()["title"], "Новое название")
        self.assertEqual(
            Video.objects.values().get(pk=self.video.pk), {**before_video, "title": "Новое название"},
        )
        self.assertEqual(Upload.objects.values().get(pk=self.upload.pk), before_upload)
        self.assertEqual(Job.objects.values().get(pk=self.job.pk), before_job)
        self.assertEqual(Outbox.objects.values().get(pk=self.outbox.pk), before_outbox)
        self.assertEqual((Video.objects.count(), Job.objects.count(), Outbox.objects.count()), (1, 1, 1))
        self.assertEqual(self.client.get("/api/videos").json()["videos"][0]["title"], "Новое название")

    def test_other_author_cannot_rename_or_list_video(self):
        self.login(self.other)
        self.assertEqual(self.client.get("/api/videos").json(), {"videos": []})
        self.assertEqual(self.rename("Чужое").status_code, 404)
        self.video.refresh_from_db()
        self.assertEqual(self.video.title, "Прежнее")

    def test_requires_session_and_csrf(self):
        response = self.client.post(self.url, {"title": "Без CSRF"}, content_type="application/json")
        self.assertEqual(response.status_code, 403)
        self.client.logout()

        self.assertEqual(self.rename("Без входа").status_code, 401)
        self.video.refresh_from_db()
        self.assertEqual(self.video.title, "Прежнее")

from django.db import transaction
from django.http import JsonResponse
from django.shortcuts import get_object_or_404
from django.views.decorators.cache import never_cache
from django.views.decorators.http import require_POST

from .accounts import error, json_object, require_author
from .models import Video


@require_POST
@never_cache
@require_author
def rename(request, video_id):
    with transaction.atomic():
        video = get_object_or_404(
            Video.objects.select_for_update(), id=video_id, owner=request.user,
        )
        title = json_object(request).get("title")
        if not isinstance(title, str) or not 1 <= len(title.strip()) <= 200 or "\0" in title:
            return error("Введите название длиной от 1 до 200 символов.", 400)
        video.title = title.strip()
        video.save(update_fields=["title"])
    return JsonResponse({"id": video.id, "title": video.title, "state": video.state})

# S05 · Переименование и права владельца

12 ноября 2026 года. Ветка `feature/s05-rename`, база `origin/main` / `f120ac2` (S03). Проверена самостоятельная часть T501; весь этап S05 не завершён. [План](../../implementation-plan-s05.md).

## Изменение

`POST /api/videos/{video_id}/rename`, JSON `{"title":"Новое название"}`. Требуются сессия владельца и CSRF. Успех — 200 с `id`, `title`, `state`; неверное название — 400, отсутствие сессии — 401, CSRF — 403, чужой или отсутствующий ID — 404. Разрешено 1–200 символов после обрезки пробелов, NUL запрещён. Меняется только Video.title под блокировкой строки, без миграций, новой обработки и изменения исходной Upload.

Кабинет позволяет открыть форму, сохранить или отменить изменение. Ошибка сохраняет введённое значение для повтора. Название выводится текстом.

## S03

Проверены три попытки с паузами 30 и 120 секунд, оба пути доставки в очередь ошибок, повторная выдача при сбое worker, heartbeat и таймауте ack. При обрыве PostgreSQL подтверждение не отправлено, после восстановления задание продолжилось. Блокировка общего тома исключила одновременную запись двух процессов FFmpeg. В обычном Compose relay доставляет задания, а обработчик с тестовым результатом запускается отдельно на тестовом стенде. [Протокол T302–T304](protocols/s03-t302-t304.md).

Регрессия локального Compose прошла. Chrome проверил путь двух авторов до принятого видео. После запуска relay в очереди было два задания и ноль потребителей. При остановке каждой зависимости live оставался 200, ready менялся с 200 на 503 и обратно. Все 8 тестов Django прошли. [Протокол регрессии](protocols/s03-t302-t304.md#регрессия-s01-и-s02).

Регрессия T304 подтвердила, что после длительного сбоя PostgreSQL тот же worker возобновляет получение заданий. Три разрыва канала и таймаут ожидания блокировки не создали попыток. Попытку начала только действующая доставка. [Протокол](protocols/t304-recovery-regression.md).

- После S04 проверить новое название в каталоге, поиске и странице просмотра, прежнюю ссылку и воспроизведение. Под общей блокировкой строки Video отклонять переименование после DELETING; согласовать порядок с удалением.

## T501 · Ручной повтор из оригинала

- Разрешать повтор владельцу после окончательного устранимого сбоя. Отказ для повреждённого или неподдерживаемого оригинала объясняет необходимость загрузить исправный файл. Не принимать повтор для активного, готового или удаляемого видео.
- **Привязать запрос к конкретному неудачному job_id**, а не только к video_id и текущему состоянию. Сохранить устойчивую связь исходного неудачного задания с созданным заданием следующего поколения; защитить связь уникальным ограничением БД.
- После проверки владельца сначала искать уже принятый повтор для этого исходного job_id. Если он существует, вернуть именно созданное задание и его текущее состояние, даже если оно уже завершилось новой ошибкой. Старый запрос не создаёт следующее поколение. Для повтора новой ошибки автор явно адресует новый failed job_id.
- Проверить оригинал под файловой защитой S04, без долгой транзакции на время чтения/хэширования. В короткой транзакции повторно проверить состояние и поколение, создать новое поколение, job, связь повтора и outbox атомарно. Сохранить ID видео, URL, байты оригинала и историю прежних попыток.
- Два параллельных запроса одного исходного job_id возвращают одно задание. Повтор публикации сообщения сохраняет message_id. Недоступность RabbitMQ не теряет принятое обязательство из outbox.
- Обработку выполнить механизмом S03–S04. Старое поколение не получает право писать или публиковать. Бюджет попыток нового задания ведётся отдельно.
- В кабинете показать допустимость повтора и переходы состояний. Сбой доставки уведомления не разрешает повторную конвертацию.

Обязательные проверки:

1. Вызвать устранимый сбой, исправить причину, повторить и получить READY без передачи файла. Сверить ID видео, URL и SHA-256 оригинала.
2. Дважды и параллельно отправить запрос с одним failed job_id: одно новое поколение, один job и одно обязательство outbox.
3. Потерять ответ первого запроса. Дождаться, пока созданное им задание тоже завершится FAILED. Повторить **старый** запрос: возвращён тот же job, новых поколения и outbox нет. Явный запрос с новым failed job_id может создать следующее поколение.
4. Проверить чужой запрос, удаление одновременно с повтором, старое сообщение, недоступный брокер и повреждённый оригинал.

## T502 · Отзыв доступа до ожидания FFmpeg

**Транзакция удаления не ждёт FFmpeg и не ждёт файловую блокировку S04.** Под короткой блокировкой строки Video проверить владельца, записать устойчивую отметку удаления, установить DELETING, отозвать поколение и сохранить обязательство очистки. Отменить ожидающие уведомления, когда их модель появится в S06. Повтор удаления безопасен.

Только **после коммита** остановить группу процессов FFmpeg и дождаться общей блокировки S04 для уборки. Новый запрос к странице, каталогу, master, вариантам и сегментам уже должен отклоняться, даже если процесс ещё завершает работу или блокировка занята. Проверка разрешения не должна ждать блокировку уборки. При недоступной БД выдача закрывается.

Остановка и уборка:

- Отправить сигнал группе FFmpeg, после ограниченного ожидания — принудительное завершение. Доказать отсутствие писателя; при неопределённости не запускать конкурентную уборку и сохранить диагностируемый DELETING.
- Под общей стабильной блокировкой S04 повторно проверить удаление и убрать оригинал, временные файлы, processing и ready. Не удалять активный lock-файл и не менять порядок блокировок вопреки S04.
- Отсутствующие файлы считать нормальным результатом повторной уборки. При ошибке оставить доступ закрытым, обязательство сохранённым и выполнить повтор после исправления причины.
- Перевести в DELETED только после завершения очистки. Историю и отметку удаления сохранить.
- Добавить подтверждение удаления в кабинете, понятный результат и состояние ожидания физической очистки.

Обязательная проверка границы: удержать общую файловую блокировку отдельным процессом, удалить ролик и подтвердить коммит DELETING. **Пока блокировка ещё удерживается**, проверить отказ новых запросов по ранее полученным ссылкам страницы, master и сегментов и отсутствие в каталоге. Затем освободить lock и проверить очистку. Уже скачанные/буферизованные байты отозвать нельзя.

Дополнительно проверить чужое удаление, удаление ожидающего/ошибочного/готового ролика, остановку записи FFmpeg и отсутствие её возобновления, прерывание и повтор уборки, ошибку прав или тома без ложного DELETED.

## T503 · Гонки, перезапуски и граница журнала

- Удалить во время FFmpeg и между rename готового каталога и коммитом READY. Любая сериализация оставляет удалённое видео закрытым; поздний результат не восстанавливает READY.
- Состязать удаление с ручным повтором и переименованием. После удаления не появляется доступный ролик или новый действующий запуск.
- Повторно доставить старую команду после удаления: она не запускает FFmpeg и не публикует результат.
- Остановить API после коммита удаления до ответа; повторить запрос. Остановить уборщик в середине очистки и перезапустить сервис. Проверить сохранность отметки в БД, закрытый доступ и завершение уборки по устойчивому обязательству.
- **Граница S05:** записать устойчивую отметку удаления и доказать её сохранение после перезапуска. Этого недостаточно для защиты от восстановления старой БД.
- **Обязательство S08:** журнал должен пережить восстановление старой БД и содержать удаления после даты backup. Согласовать независимое хранение/копирование, монотонную отметку, retention и сверку до открытия выдачи. При неполном журнале не открывать восстановленные ролики до ручной сверки. Проверку на старой резервной копии выполнить в **T802**.
- **После T602:** удалить при ожидающем и уже начатом уведомлении. Ожидающее отменить; уже начатый внешний эффект гарантированно отменить нельзя. Записанная строка остаётся историей, её ссылка закрыта. Поздний ответ не меняет удалённое видео и не запускает обработку.

"""Check S05 rename in a running disposable Compose project; leaves labelled fixtures."""

import os
import secrets
import shutil
import subprocess
import uuid

from playwright.sync_api import expect, sync_playwright

from smoke_s02 import BASE_URL, COMPOSE, register_and_sign_in


def main():
    suffix = uuid.uuid4().hex[:12]
    owner, other = f"s05_owner_{suffix}", f"s05_other_{suffix}"
    password = secrets.token_urlsafe(24)
    chrome = os.getenv("VIDHARBOR_CHROME") or shutil.which("chromium") or shutil.which("google-chrome")
    with sync_playwright() as playwright:
        browser = playwright.chromium.launch(executable_path=chrome, args=["--no-sandbox"])
        owner_context = browser.new_context()
        page = owner_context.new_page()
        register_and_sign_in(page, owner, password)
        # S04 publication is not available yet. Seed a QUEUED video to exercise real rename/list APIs.
        code = (
            "from django.contrib.auth import get_user_model; "
            "from vidharbor.models import Upload, Video; "
            f"owner = get_user_model().objects.get(username={owner!r}); "
            "upload = Upload.objects.create(owner=owner, title='S05 fixture', expected_size=1, "
            "state=Upload.State.COMPLETE); "
            "video = Video.objects.create(owner=owner, upload=upload, title='S05 fixture'); "
            "print(video.id)"
        )
        video_id = subprocess.check_output(
            [*COMPOSE, "api", "python", "manage.py", "shell", "-c", code], text=True, timeout=30,
        ).strip().splitlines()[-1]
        uuid.UUID(video_id)
        page.reload()
        row = page.get_by_role("listitem").filter(has_text="S05 fixture")
        row.get_by_role("button", name="Переименовать", exact=True).click()
        page.get_by_label("Новое название", exact=True).fill("Не сохранять")
        page.get_by_role("button", name="Отмена", exact=True).click()
        expect(row.locator("strong")).to_have_text("S05 fixture")
        row.get_by_role("button", name="Переименовать", exact=True).click()
        title = '<img src=x onerror="window.s05Injected=true">'
        page.get_by_label("Новое название", exact=True).fill(title)
        route_pattern = "**/api/videos/*/rename"
        page.route(route_pattern, lambda route: route.fulfill(
            status=503, content_type="application/json", body='{"error":"Тестовый отказ"}',
        ), times=1)
        page.get_by_role("button", name="Сохранить", exact=True).click()
        expect(page.get_by_role("alert")).to_have_text("Тестовый отказ")
        expect(page.get_by_label("Новое название", exact=True)).to_have_value(title)
        page.get_by_role("button", name="Сохранить", exact=True).click()
        expect(page.get_by_role("status")).to_have_text("Название сохранено")
        expect(page.locator(".videos strong")).to_have_text(title)
        page.reload()
        expect(page.locator(".videos strong")).to_have_text(title)
        assert not page.evaluate("Boolean(window.s05Injected)")
        expect(page.locator(".videos img")).to_have_count(0)

        other_context = browser.new_context()
        other_page = other_context.new_page()
        register_and_sign_in(other_page, other, password)
        expect(other_page.get_by_role("listitem")).to_have_count(0)
        path = f"/api/videos/{video_id}/rename"
        token = other_context.request.get(BASE_URL + "/api/auth/session").json()["csrf_token"]
        denied = other_context.request.post(
            BASE_URL + path, data={"title": "Чужое"}, headers={"X-CSRFToken": token},
        )
        assert denied.status == 404, denied.status
        assert owner_context.request.post(BASE_URL + path, data={"title": "Без CSRF"}).status == 403
        page.reload()
        expect(page.locator(".videos strong")).to_have_text(title)
        browser.close()
    print("S05 rename: save, reload, cancel, error retry, HTML escaping, owner isolation and CSRF passed.")
    print("QUEUED fixture only; catalogue, search, playback and deletion require S04.")


if __name__ == "__main__":
    main()

import { createApp, h, onMounted, ref } from "vue";

import { api, ApiError, sendFile, type Session, type Video } from "./api";
import "./style.css";

function messageOf(error: unknown) {
  return error instanceof Error ? error.message : "Сервис недоступен.";
}

createApp({
  setup() {
    const session = ref<Session>();
    const videos = ref<Video[]>([]);
    const registering = ref(false);
    const busy = ref(false);
    const error = ref("");
    const status = ref("");
    const progress = ref<number>();
    const pending = ref<string>();
    const editing = ref<string>();
    const editedTitle = ref("");

    async function loadVideos() {
      videos.value = (await api<{ videos: Video[] }>("/api/videos")).videos;
    }

    async function run(action: () => Promise<void>) {
      error.value = "";
      busy.value = true;
      try {
        await action();
      } catch (failure) {
        error.value = messageOf(failure);
      } finally {
        busy.value = false;
      }
    }

    function refreshSession() {
      return run(async () => {
        session.value = await api<Session>("/api/auth/session");
        if (session.value.username) await loadVideos();
      });
    }
    onMounted(refreshSession);

    async function submitCredentials(event: Event) {
      event.preventDefault();
      if (!session.value) return;
      const form = event.currentTarget as HTMLFormElement;
      const data = Object.fromEntries(new FormData(form));
      const csrf = session.value.csrf_token;
      await run(async () => {
        status.value = "";
        if (registering.value) {
          await api("/api/auth/register", { csrf, data });
          status.value = "Аккаунт создан. Теперь войдите.";
          registering.value = false;
        } else {
          session.value = await api<Session>("/api/auth/login", { csrf, data });
          await loadVideos();
        }
        form.reset();
      });
    }

    async function completeUpload(csrf: string) {
      if (!pending.value) return;
      status.value = "Проверяем файл";
      try {
        await api("/api/uploads/" + pending.value + "/complete", { csrf });
      } catch (failure) {
        if (failure instanceof ApiError && failure.status < 500 && !failure.retryAfter) {
          pending.value = undefined;
          status.value = "";
        } else {
          status.value = "Проверку можно повторить";
        }
        throw failure;
      }
      pending.value = undefined;
      status.value = "Видео принято и ожидает обработки";
    }

    async function submitUpload(event: Event) {
      event.preventDefault();
      if (!session.value || pending.value) return;
      const form = event.currentTarget as HTMLFormElement;
      const data = new FormData(form);
      const file = data.get("file");
      if (!(file instanceof File) || !file.size) {
        error.value = "Выберите непустой файл.";
        return;
      }
      const csrf = session.value.csrf_token;
      await run(async () => {
        progress.value = 0;
        status.value = "Передаём файл";
        try {
          const upload = await api<{ id: string }>("/api/uploads", {
            csrf, data: { title: data.get("title"), size: file.size },
          });
          await sendFile(upload.id, file, csrf, (percent) => { progress.value = percent; });
          progress.value = 100;
          pending.value = upload.id;
          form.reset();
        } catch (failure) {
          status.value = "";
          progress.value = undefined;
          throw failure;
        }
        await completeUpload(csrf);
        await loadVideos();
      });
    }

    function retryCompletion() {
      const csrf = session.value?.csrf_token;
      if (!csrf) return;
      return run(async () => {
        await completeUpload(csrf);
        await loadVideos();
      });
    }

    function signOut() {
      const csrf = session.value?.csrf_token;
      if (!csrf) return;
      return run(async () => {
        session.value = await api<Session>("/api/auth/logout", { csrf });
        videos.value = [];
        status.value = "";
        progress.value = undefined;
        pending.value = undefined;
        editing.value = undefined;
        editedTitle.value = "";
      });
    }

    function startRename(video: Video) {
      editing.value = video.id;
      editedTitle.value = video.title;
      error.value = "";
      status.value = "";
    }

    function submitRename(event: Event) {
      event.preventDefault();
      const csrf = session.value?.csrf_token;
      const id = editing.value;
      if (!csrf || !id || busy.value) return;
      return run(async () => {
        const updated = await api<Video>("/api/videos/" + id + "/rename", {
          csrf, data: { title: editedTitle.value },
        });
        videos.value = videos.value.map((video) => video.id === id ? updated : video);
        editing.value = undefined;
        status.value = "Название сохранено";
      });
    }

    function credentials() {
      return h("section", { "data-testid": "auth-screen", "aria-labelledby": "auth-title" }, [
        h("h2", { id: "auth-title" }, registering.value ? "Регистрация" : "Вход"),
        h("form", { onSubmit: submitCredentials }, [
          h("label", ["Имя пользователя", h("input", {
            name: "username", autocomplete: "username", required: true, maxlength: 150,
          })]),
          h("label", ["Пароль", h("input", {
            name: "password", type: "password", required: true,
            autocomplete: registering.value ? "new-password" : "current-password",
          })]),
          h("div", { class: "actions" }, [
            h("button", { type: "submit", disabled: busy.value }, registering.value ? "Зарегистрироваться" : "Войти"),
            h("button", {
              type: "button", class: "secondary", disabled: busy.value,
              onClick: () => {
                registering.value = !registering.value;
                error.value = "";
                status.value = "";
              },
            }, registering.value ? "Ко входу" : "Создать аккаунт"),
          ]),
        ]),
      ]);
    }

    function cabinet() {
      return h("section", { "aria-labelledby": "cabinet-title" }, [
        h("div", { class: "cabinet-header" }, [
          h("h2", { id: "cabinet-title" }, "Кабинет · " + session.value?.username),
          h("button", { class: "secondary", disabled: busy.value, onClick: signOut }, "Выйти"),
        ]),
        h("form", { onSubmit: submitUpload }, [
          h("label", ["Название видео", h("input", { name: "title", required: true, maxlength: 200 })]),
          h("label", ["Видеофайл", h("input", {
            name: "file", type: "file", accept: ".mp4,.mov,.mkv,.webm", required: true,
          })]),
          h("div", { class: "actions" }, [
            h("button", { type: "submit", disabled: busy.value || !!pending.value }, "Загрузить"),
            pending.value && !busy.value
              ? h("button", { type: "button", onClick: retryCompletion }, "Повторить проверку")
              : null,
          ]),
          progress.value === undefined ? null : h("div", { class: "progress" }, [
            h("progress", { value: progress.value, max: 100, "aria-label": "Прогресс загрузки" }),
            h("output", progress.value + " %"),
          ]),
        ]),
        h("h3", "Мои видео"),
        videos.value.length
          ? h("ul", { class: "videos" }, videos.value.map((video) => h("li", { key: video.id }, [
            h("strong", video.title),
            h("span", video.state === "QUEUED" ? "Ожидает обработки" : video.state),
            editing.value === video.id
              ? h("form", { onSubmit: submitRename, "aria-label": "Переименование видео" }, [
                h("label", ["Новое название", h("input", {
                  name: "title", required: true, maxlength: 200, value: editedTitle.value,
                  disabled: busy.value,
                  onInput: (event: Event) => { editedTitle.value = (event.target as HTMLInputElement).value; },
                })]),
                h("div", { class: "actions" }, [
                  h("button", { type: "submit", disabled: busy.value }, "Сохранить"),
                  h("button", {
                    type: "button", class: "secondary", disabled: busy.value,
                    onClick: () => { editing.value = undefined; error.value = ""; },
                  }, "Отмена"),
                ]),
              ])
              : h("button", {
                type: "button", class: "secondary", disabled: busy.value,
                onClick: () => startRename(video),
              }, "Переименовать"),
          ])))
          : h("p", "Здесь появятся загруженные видео."),
      ]);
    }

    return () => h("main", [
      h("h1", "VidHarbor"),
      session.value?.username ? cabinet() : session.value ? credentials() : null,
      !session.value && !busy.value
        ? h("button", { onClick: refreshSession }, "Повторить подключение")
        : null,
      status.value ? h("p", { role: "status" }, status.value) : null,
      error.value ? h("p", { role: "alert" }, error.value) : null,
    ]);
  },
}).mount("#app");
